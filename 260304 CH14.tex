\chapter{Définitions}

\begin{definition}[Espace vectoriel]
    Soit $\Kset$ un corps commutatif.

    $(E,+,\cdot)$ est dit $\Kset$-espace vectoriel (ou $\Kset$-ev.) lorsque :
    \begin{enumerate}
        \item $(E,+)$ est un groupe abélien.
        \item $\cdot : \Kset \times E \to E$ est une loi de composition externe (LCE), donc une application vérifiant :
        \begin{itemize}
            \item $\forall \lambda, \mu \in \Kset, \forall u \in E, (\lambda + \mu) \cdot u = \lambda \cdot u + \mu \cdot u$.
            \item $\forall \lambda \in \Kset, \forall u, v \in E, \lambda \cdot (u + v) = \lambda \cdot u + \lambda \cdot v$.
            \item $\forall \lambda, \mu \in \Kset, \forall u \in E, (\lambda\mu) \cdot u = \lambda\cdot(\mu\cdot u)$.
            \item $\forall u\in E, 1\cdot u = u$.
        \end{itemize}
    \end{enumerate}
\end{definition}

\begin{attention}
    Il faut faire attention à ne pas confondre les notations $+$ et $\cdot$ dans $E$ et $\Kset$.
\end{attention}

\begin{property}
    Soit $E$ un $\Kset$-ev. Alors :
    \begin{itemize}
        \item $\forall x \in E, \forall \alpha \in \Kset, \alpha \cdot x = 0_E \iff \alpha = 0_\Kset \ou x = 0_E$.
        \item $\forall x \in E, -1_\Kset \cdot x = -x$.
    \end{itemize}
\end{property}

\begin{proof}
    \begin{outline}
        \1 Démonstration de la première propriété :
            \2 Dans le sens indirect $\impliedby$, $0_\Kset \cdot x = (0_\Kset + 0_\Kset) \cdot x = 0_\Kset \cdot x + 0_\Kset \cdot x$, donc $(0_\Kset \cdot x) + (-0_\Kset \cdot x) = (0_\Kset \cdot x + 0_\Kset \cdot x) + (-0_\Kset \cdot x)$, donc $0_E = 0_\Kset \cdot x$.
            \2 On a aussi $\forall \alpha \in \Kset$, $0_E \cdot \alpha = 0_E \implies 0_E = 0_E \cdot \alpha = (0_E + 0_E) \cdot \alpha = 0_E \cdot \alpha + 0_E \cdot \alpha$, donc $0_E = 0_E \cdot \alpha$.
            \2 Dans le sens direct $\implies$, $0_E = \alpha \cdot x$
    \end{outline}
\end{proof}

\begin{remark}
    \begin{enumerate}
        \item On déduit de la première et deuxième propriété de la définition et de la deuxième propriété précédente que :
        $$\forall \alpha,\beta \in \Kset, \forall x,y \in E,\\
        \begin{cases}
        (\alpha - \beta) \cdot x = \alpha \cdot x - \beta \cdot x\\
        \alpha \cdot (x - y) = \alpha \cdot x - \alpha \cdot y
        \end{cases}$$
        \item Les éléments de $E$ sont appelés des vecteurs, les éléments de $\Kset$ sont appelés des scalaires.
    \end{enumerate}
\end{remark}

\begin{example}
    \begin{enumerate}
        \item Si $\Kset$ est un corps commutatif, alors $\Kset^n$ avec $n\in \N^*$ est un $\Kset$-ev avec les lois de composition suivantes :
        \begin{itemize}
            \item $\forall u=(u_1,\dots,u_n), v=(v_1,\dots,v_n) \in \Kset^n, u+v = (u_1 + v_1, \dots, u_n + v_n)$.
            \item $\forall \lambda \in \Kset, \forall u=(u_1,\dots,u_n) \in \Kset^n, \lambda\cdot u = (\lambda\cdot u_1, \dots, \lambda\cdot u_n)$.
        \end{itemize}
        \item Si $E$ est un corps et $\Kset$ est un sous-corps commutatif de $E$, alors $E$ est un $\Kset$-ev dont la loi de composition externe est la multiplication de $E$. Par exemple, $\C$ est un $\R$-ev.
        $$\Kset\times E \to E, (\lambda, x) \mapsto \lambda\cdot x = \lambda x$$
        \item Si $\Kset$ est un corps commutatif et $D$ est un ensemble non-vide, alors $(\Kset^D, +, \cdot)$ est un $\Kset$-ev, avec :
        $$\forall x,y \in \Kset^D, \forall \lambda \in \Kset,\\
        \begin{cases}
        x+y:\begin{cases}
        D\to\Kset\\
        t\mapsto x(t)+y(t)
        \end{cases}\\
        \lambda\cdot x:\begin{cases}
        D\to\Kset\\
        t\mapsto \lambda\cdot x(t)
        \end{cases}
        \end{cases}$$
    \end{enumerate}
\end{example}

\chapter{Sous-espaces vectoriels}

\begin{definition}
    Soit $E$ un $\Kset$-ev, et soit $F$ une partie de $E$.

    On dit que $F$ est un sous-espace vectoriel (sous-ev.) de $E$ lorsque $F$ est lui-même un $\Kset$-ev pour les lois de composition induites par celles de $E$.
\end{definition}

\section{Caractérisation d'un sous-espace vectoriel}

\begin{property}[Caractérisation d'un sous-espace vectoriel]
    Soit $E$ un $\Kset$-ev, et soit $F$ une partie de $E$.

    Les propriétés suivantes sont équivalentes :
    \begin{enumerate}
        \item $F$ est un sous-ev. de $E$.
        \item $F \neq \emptyset$, et pour tous $x,y\in F$ et $\alpha,\beta \in\Kset$, on a $\alpha x + \beta y \in F$.
        \item $0_E \in F$, et pour tous $x,y\in F$ et $\lambda \in\Kset$, on a $\lambda x + y \in F$ (très pratique pour montrer que $F$ est un sous-ev. de $E$).
    \end{enumerate}
\end{property}

\begin{proof}
    Pour démontrer ces équivalences, on va suivre un raisonnement circulaire classique en montrant que $(1) \implies (2)$, $(2) \implies (3)$, et $(3) \implies (1)$.

    \begin{outline}
        \1 Montrons que $(1) \implies (2)$ :
            \2 On a $F$ est un $\Kset$-ev., alors $(F,+)$ est un groupe abélien, donc $F \neq \empty$.
            \2 Donc $\forall \alpha,\beta \in \Kset,\forall x,y \in F, \begin{cases}
            \alpha x \in F\\
            \beta y \in F
            \end{cases}$
            \2 Donc $\alpha x + \beta y \in F$.
        \1 Montrons que $(2) \implies (3)$ :
            \2 On sait que $F \neq \emptyset$.
            \2 Donc pour $\alpha = \beta = 0_\Kset$, et $y = x \in F$, on a $0_E = 0_\Kset \cdot x + 0_\Kset \cdot x \in F$.
            \2 Donc $\forall \alpha \beta \text{ qu'on fixe sur } (1_\Kset,\lambda) \in\Kset^2, \forall x,y \in F, \lambda x + y \in F$.
        \1 Montrons que $(3) \implies (1)$ :
            \2 $0_E \in F$, et pour tous $x,y \in F, x-y \in F$ (en prenant $\lambda = -1_\Kset$).
            \2 Donc $F$ est un sous-groupe de $(E,+)$.
            \2 Donc $(F,+)$ est un groupe abélien.
            \2 On a pour $x = 0_E$, et pour tous $\lambda, y \in \Kset\times F, 0_E + \lambda \cdot y \in F$, donc $\lambda \cdot y \in F$.
            \2 Donc $\cdot$ est une loi de composition externe de $F$.
            \2 Les propriétés de la loi de composition externe sont vérifiées dans $F$ car elles sont vérifiées dans $E$.
            \2 Donc $F$ est un $\Kset$-ev.
    \end{outline}
\end{proof}

\begin{example}
    \begin{outline}
        \1 Soient $E$ un $\Kset$-ev. et $a \in E\setminus\{0_E\}$. L'ensemble $\Kset \cdot a = \{\lambda \cdot a \mid \lambda \in \Kset\}$ est appelé la droite vectorielle engendrée par $a$, et est un sous-ev. de $E$.
        \1 $F = \{(x,y) \in \R^2 \mid xy = 0\}$ n'est pas un sous-ev. de $\R^2$, parce que $(1,0)$ et $(0,1)$ sont dans $F$, mais pas leur somme $(1,1)$.
        \1 Soit $a = (a_1,\dots,a_n) \in \Kset^n$. On pose $H = \{x=(x_1,\dots,x_n) \in \Kset^n \mid \sum_{i=1}^{n} a_i x_i = 0\}$. Alors $H$ est un sous-ev. de $\Kset^n$, appelé l'hyperplan de $\Kset^n$ d'équation $\sum_{i=1}^n a_i x_i = 0$.
            \2 On a bien $H \subset \Kset^n$.
            \2 On a $\sum_{i=1}^{n}a_i \cdot 0_\Kset = 0_\Kset$, donc $0_{\Kset^n} \in H$.
            \2 Soient $x=(x_1,\dots,x_n), y=(y_1,\dots,y_n) \in H$, et $\lambda \in \Kset$. Alors $\sum_{i=1}^{n} a_i (\lambda x_i + y_i) = \lambda \sum_{i=1}^{n} a_i x_i + \sum_{i=1}^{n} a_i y_i = 0_\Kset$, donc $\lambda x + y \in H$.
            \2 Et on a bien $\sum_{i=1}^{n} a_i (\lambda x_i + y_i) = \lambda \sum_{i=1}^{n} a_i x_i + \sum_{i=1}^{n} a_i y_i = 0_\Kset$, donc $\lambda x + y \in H$.
            \2 Donc $H$ est un sous-ev. de $\Kset^n$.
        \1 $\cclas{n}(I,\Kset)$ est un sous-ev. de $\Kset^I$.
        \1 $\Kset[X]$ est un sous-ev. de $\Kset^{\N}$.
        \1 $\Kset_n[X]$ est un sous-ev. de $\Kset[X]$.
    \end{outline}
\end{example}

\begin{property}
    Soit $E$ un $\Kset$-ev.
    \begin{enumerate}
        \item Si $(F_i)_{i\in I}$ est une famille de sous-ev. de $E$, alors $\bigcap_{i\in I} F_i$ est un sous-ev. de $E$.
        \item Soient $F,G$ deux sous-ev. de $E$. Alors $F \cup G$ est un sous-ev. de $E$ si et seulement si $F \subset G$ ou $G \subset F$.
    \end{enumerate}
\end{property}

\begin{proof}
    \begin{outline}
        \1 Démonstration de la première propriété...
            \2 \textbf{Inclusion : } $\bigcap_{i \in I} F_i \subset E$.
            \2 On sait aussi que $\forall i \in I, 0_E \in F_i \implies 0_E \in \bigcap_{i\in I} F_i$.
            \2 Soient $x,y \in \bigcap_{i\in I} F_i$, et $\lambda \in \Kset$. Alors $\forall i \in I, x,y \in F_i$, donc $\lambda x + y \in F_i$, donc $\lambda x + y \in \bigcap_{i\in I} F_i$, sachant que $F_i$ est un sous-ev. de $E$.
            \2 On en déduit que $\bigcap_{i\in I} F_i$ est un sous-ev. de $E$.
        \1 Démonstration de la deuxième propriété...
            \2 Dans le sens indirect $\impliedby$, si $F \subset G$, alors $F \cup G = G$ est un sous-ev. de $E$. De même, si $G \subset F$, alors $F \cup G = F$ est un sous-ev. de $E$.
            \2 Dans le sens direct $\implies$, on utilise la définition d'un sous-espace vectoriel.
            \2 En effet, comme $F \cup G$ est un sous-ev. de $E$, alors $F \cup G$ est un sous-groupe de $(E,+)$.
            \2 D'après les propriétés des groupes, on a bien $F \subset G$ ou $G \subset F$.
    \end{outline}
\end{proof}

\section{Sous-espace vectoriel engendré}

\begin{definition}[Sous-espace vectoriel engendré]
    Soient $E$ un $\Kset$-ev. et $A$ une partie de $E$.

    $\bigcap_{\substack{F \text{ sev. de } E\\ A \subset F}} F$ est appelé \textbf{sous-espace vectoriel engendré} par $A$, et est noté $\Vect(A)$.
\end{definition}

\begin{property}
    Il s'agit du plus petit sous-ev. de $E$ contenant $A$.
\end{property}

\begin{remark}
    \begin{enumerate}
        \item Si $F = \Vect(A)$, alors on dit que $A$ est une partie génératrice de $F$.
        \item $\Vect(\emptyset) = \{0_E\}$, et $\Vect(\{x\}) = \Kset \cdot x$.
        \item \textbf{Important :} Soit $E$ un $\Kset$-ev. et $A$ une partie non-vide de $E$.\\$x \in \Vect(A) \iff \exists n \in \N^*, \exists a_1,\dots,a_n \in A, \exists \lambda_1,\dots,\lambda_n \in \Kset \mid x = \sum_{i=1}^n \lambda_i a_i$\\Le terme $\sum_{i=1}^n \lambda_i a_i$ est appelé une combinaison linéaire de $a_1,\dots,a_n$.\\i.e. $\Vect(A) = \{ \sum_{i=1}^{n}\lambda_i a_i \tq n \in \N^*, a_1,\dots,a_n \in A, \lambda_1,\dots,\lambda_n \in \Kset\}$.
    \end{enumerate}
\end{remark}

\begin{example}
    \begin{outline}[enumerate]
        \1 Si $A = \{a_1,\dots,a_n\}$, alors $\Vect(A)$ est noté $\Vect(a_1,\dots,a_n)$, et est égal à $\{\sum_{i=1}^n \lambda_i a_i \tq \lambda_1,\dots,\lambda_n \in \Kset\}$.
        \1 Soit $F = \{(x,y,z) \in \R^3 \tqp x-2y+z=0\}$.
        \2 On a $F = \{(2y - z, y, z) \tqp y,z \in \R\}$.
        \2 Donc $F = \{y(2,1,0) + z(-1,0,1) \tq y,z \in \R\}$.
        \2 Donc $F = \Vect((2,1,0), (-1,0,1))$.
        \1 On a $\Kset^n = \{x = (x_1,\dots,x_n) \tq x_1,\dots,x_n \in \Kset\} = \{\sum_{i=1}^n x_i e_i \tq x_1,\dots,x_n \in \Kset\}$, où $e_i$ est le vecteur de $\Kset^n$ dont la $i$-ème composante est égale à $1_\Kset$, et les autres composantes sont égales à $0_\Kset$ (donc $e_k = (0,\dots,\underbrace{0}_{\text{pos. } k},0,\dots,0)$).
        \1 $\Kset[X] = \Vect((X^k)_{k \in \N})$.
        \2 En effet, $\exists d \in \N, \exists \alpha_0,\dots,\alpha_d \in \Kset \tq P = \sum_{k=0}^{d}\alpha_k X^k$.
        \1 $\Kset_n[X] = \Vect(1,X,\dots,X^n) = \Vect((X^k)_{0 \leq k \leq n})$.
        \1 $F = \{ f \in \cclas\infty(\R,\R) \tqf f'' + 2f' - 3f = 0\}$
        \2 Donc $F = \{f:t\mapsto \alpha e^t + \beta e^{-t} \tq \alpha,\beta \in \R\}$.
        \2 Donc $F = \Vect(e^t, e^{-t})$.
    \end{outline}
\end{example}

\begin{property}
    Soient $E$ un $\Kset$-ev. et $A$,$B$ deux parties de $E$. Alors :
    \begin{enumerate}
        \item $\Vect(A) = A \iff A \text{ est un sous-ev. de } E$.
        \item $\Vect(\Vect(A)) = \Vect(A)$.
        \item $A \subset B \implies \Vect(A) \subset \Vect(B)$.
        \item $\Vect(A) = \Vect(B) \iff \begin{cases}
        A \subset \Vect(B)\\
        B \subset \Vect(A)
        \end{cases}$.
    \end{enumerate}
\end{property}

\begin{proof}
    \begin{outline}
        \1 Démonstration de la quatrième propriété.
            \2 Dans le sens direct $(\implies)$, $A \subset \Vect(A) = \Vect(B)$, et $B \subset \Vect(B) = \Vect(A)$.
            \2 Dans le sens indirect $(\impliedby)$
                \3 $A \subset \Vect(B) \overset{\text{(3)}}\implies \Vect(A) \subset \Vect(\Vect(B)) \overset{\text{(2)}}\implies \Vect(A) \subset \Vect(B)$,
                \3 et $B \subset \Vect(A) \overset{\text{(3),(2)}}\implies \Vect(B) \subset \Vect(A)$, donc $\Vect(A) = \Vect(B)$.
    \end{outline}
\end{proof}

\chapter{Famille génératrice, libre et bases}

\section{Famille génératrice}

\begin{definition}
    Soient $E$ un $\Kset$-ev. et $(e_i)_{i \in I}$ une famille d'éléments de $E$.

    On dit que $(e_i)_{i \in I}$ est une famille génératrice de $E$ lorsque $\Vect(e_i, i \in I) = E$.

    i.e. lorsque $\forall x \in E, \exists J \text{ sous-ensemble fini de } I, \exists (\lambda_j)_{j \in J} \in \Kset^{\card{J}} \tq x = \sum_{j \in J} \lambda_j e_j$.

    \textit{Note : on ne peut pas définir cette famille génératrice à partir de la combinaison linéaire de tous les éléments de la famille, car il peut y en avoir une infinité.}
\end{definition}

\begin{remark}
    Si $I$ est fini tel que $\card{I} = p \in \N^*$, alors :
    $$(e_i)_{i \in I} \text{ famille génératrice de } E \iff \forall x \in E, \exists (\lambda_k)_{k\in I} \in \Kset^p \tqp x = \sum_{k \in I}^p \lambda_k e_k$$

    Si $I = \{1,\dots,p\}$, alors :
    $$(e_1,\dots,e_p) \text{ famille génératrice de } E \iff \forall x \in E, \exists (\lambda_1,\dots,\lambda_p) \in \Kset^p \tq x = \sum_{k=1}^p \lambda_k e_k$$
\end{remark}

\begin{example}
    \begin{enumerate}
        \item $\Kset^n = \Vect(e_1,\dots,e_n)$, donc $(e_1,\dots,e_n)$ est une famille génératrice de $\Kset^n$.
        \item $\Kset[X] = \Vect((X^k)_{k \in \N})$, donc $((X^k)_{k \in \N})$ est une famille génératrice de $\Kset[X]$.
        \item $\Kset_n[X] = \Vect(1,X,\dots,X^n)$, donc $(1,X,\dots,X^n)$ est une famille génératrice de $\Kset_n[X]$.
    \end{enumerate}
\end{example}

\begin{property}
    Toute surfamille d'une famille génératrice est une famille génératrice.
    
    i.e. si $(e_i)_{i \in I}$ est une famille génératrice de $E$, et $J$ est un sous-ensemble de $I$ tel que $I \subset I'$, alors $(e_i)_{i \in I'}$ est une famille génératrice de $E$.
\end{property}

\begin{proof}
    On a $\forall x \in E, \exists J \subset I \text{ fini}, \exists (\lambda_j)_{j \in J} \in \Kset^\card{J} \tq x = \sum_{j \in J} \lambda_j e_j$, or $I \subset I'$, donc $J$ est aussi un sous-ensemble de $I'$, donc $(e_i)_{i \in I'}$ est une famille génératrice de $E$.
\end{proof}

\begin{exercise}
    Montrer que $F = \{ P \in \R_3[X] \tq \int_{0}^{1} P(t) dt = 0\}$ est un sous-ev. de $\R_3[X]$, puis trouver une famille génératrice de $F$.
\end{exercise}

\section{Famille libre}

\begin{definition}[Famille libre et famille liée]
    Soit $E$ un $\Kset$-ev.
    \begin{enumerate}
        \item Une famille $(e_i)_{1 \le i \le n}$ d'éléments de $E$ est dite \textbf{libre} lorsque :
        $$\forall (\lambda_1,\dots,\lambda_n) \in \Kset^n, \sum_{i=1}^n \lambda_i e_i = 0_E \implies \forall i \in \{1,\dots,n\}, \lambda_i = 0_\Kset$$
        \\Dans le cas contraire, la famille est dite liée.
        \item \textbf{S'il y a une infinité d'éléments}, Une famille $(e_i)_{i \in I}$ d'éléments de $E$ est dite libre lorsque :
        $$\forall J \text{ sous-ensemble fini de } I, \forall (\lambda_j)_{j \in J} \in \Kset^{\card{J}}, \sum_{j \in J} \lambda_j e_j = 0_E \implies \forall j \in J, \lambda_j = 0_\Kset$$
    \end{enumerate}
\end{definition}

\begin{remark}
    \begin{enumerate}
        \item Toute famille qui contient le $0_E$ est liée, car $1_\Kset \cdot 0_E = 0_E$.
        \item $(e_i)_{1 \le i \le n}$ est liée si et seulement si $\exists \lambda_1,\dots,\lambda_n \in \Kset$, pas tous nuls, tels que $\sum_{i=1}^n \lambda_i e_i = 0_E$. Cela équivaut aussi à dire que $\exists i_0 \in \{1,\dots,n\}$ tel que $e_{i_0} \in \Vect(e_1,\dots,e_{i_0-1}, e_{i_0+1},\dots,e_n)$.
    \end{enumerate}
\end{remark}

\begin{consequence}
    On a $\exists (\lambda_1,\dots,\lambda_n) \in \Kset^\N\setminus\{0_\Kset^\N\} \tq \sum_{k=1}^n \lambda_k e_k = 0$.

    Donc $\exists i_0 \in \{1,\dots,n\} \tq \lambda_{i_0} \neq 0$, alors :
    
    $$e_{i_0} = -\lambda_{i_0}^{-1} \sum_{\substack{k=1\\k\neq i_0}}^n \lambda_k e_k \in \Vect(e_1,\dots,e_{i_0-1}, e_{i_0+1},\dots,e_n)$$

\end{consequence}

\begin{example}
    \begin{enumerate}
        \item Dans $\Kset^n$, la famille $(e_1,\dots,e_n)$ est libre, car :
        $$
            \forall \lambda_1,\dots,\lambda_n \in \Kset, \sum_{k=1}^n \lambda_k e_k = 0_E \implies \forall k \in \{1,\dots,n\}, \lambda_k = 0_\Kset
        $$
        \item $(X^k)_{k \in \N}$ est une famille libre de $\Kset[X]$.
        \begin{itemize}
            \item Soit $J$ un sous-ensemble fini de $\N$, par exemple $J = \{0,\dots,d\} \tq d \in \N$.
            \item Soient $\lambda_0,\dots,\lambda_d \in \Kset, \tqp \sum_{k=0}^d \lambda_k X^k = 0_E \implies \forall k \in \{0,\dots,d\}, \lambda_k = 0_\Kset$ (car les polynômes sont égaux si et seulement si leurs coefficients sont égaux)
        \end{itemize}
        \item $(X^k)_{0 \le k \le n}$ est une famille libre de $\Kset_n[X]$.
        \item Soient $x_0,\dots,x_n \in \Kset$ deux à deux distincts (pour simplifier les calculs).\\Soit $l_k = \prod_{\substack{j=0\\j\neq k}}^n \frac{X-x_j}{x_k - x_j}$ le polynôme de Lagrange associé à $x_k$. Alors $(l_0,\dots,l_n)$ est une famille libre de $\Kset_n[X]$.
        $$
            \text{Soit } (\lambda_0,\dots,\lambda_n) \in \Kset^{n+1}, \tqp \sum_{k=0}^n \lambda_k l_k = 0_E\\
            \forall i \in \{0,\dots,n\}, \sum_{k=0}^n \lambda_k \underbrace{l_k(x_i)}_{= \delta_{i,k}} = \lambda_i = 0_\Kset
        $$
    \end{enumerate}
\end{example}

\begin{exercise}
    Soient $P_0,\dots,P_r \in \Kset_n[X]$ tels que $r \le n$, et $0 \le \deg(P_0) < \dots < deg(P_r) \le n$. 
    
    La famille $(P_0,\dots,P_r)$ est dite famille échelonnée dans $\Kset_n[X]$.
    
    Montrer que $(P_0,\dots,P_r)$ est une famille libre de $\Kset_n[X]$.

    \textbf{Solution :}

    Nous allons procéder par récurrence inverse sur $k \in \{0,\dots,r\}$.

    Soient $\lambda_0,\dots,\lambda_r \in \Kset$ tels que $\sum_{k=0}^r \lambda_k P_k = 0$.

    Donc $\lambda_r P_r = -\sum_{k=0}^{r-1} \lambda_k P_k$.

    Si $\lambda_r \ne 0$, alors :

    $$\deg(\lambda_r P_r) = \deg(P_r) \et \deg(-\sum_{k=0}^{r-1} \lambda_k P_k) \le \max_{0 \le k \le r-1} \deg(P_k) = \deg(P_{r-1}) < \deg(P_r)$$

    Ce qui est absurde, donc $\lambda_r = 0$.
    
    Donc $\sum_{k=0}^{r-1} \lambda_k P_k = 0$.

    D'après le principe de récurrence, on a $\lambda_{r-1} = 0$, puis $\lambda_{r-2} = 0$, et ainsi de suite jusqu'à $\lambda_0 = 0$.

    \textbf{Solution plus rigoureuse :}

    Nous allons raisonner par absurde.

    \begin{outline}
        \1 Supposons que $(P_0,\dots,P_r)$ n'est pas une famille libre de $\Kset_n[X]$.
        \1 Donc $\exists (\lambda_0,\dots,\lambda_r) \in \Kset^{r+1}\setminus\{0_\Kset^{r+1}\} \tq \sum_{k=0}^r \lambda_k P_k = 0$.
        \1 On pose $s = \max(\{ k \in \{0,\dots,r\} \tq \lambda_k \ne 0\})$.
        \1 Donc $\lambda_s P_s = -\sum_{k=0}^{s-1} \lambda_k P_k$ et $\forall k > s, \lambda_k = 0$.
        \1 Donc $\sum_{k=0}^s \lambda_k P_k = 0$.
        \1 Donc $\lambda_s P_s = -\sum_{k=0}^{s-1} \lambda_k P_k$.
        \1 Donc $\deg(\lambda_s P_s) = \deg(P_s)$, et $\deg(-\sum_{k=0}^{s-1} \lambda_k P_k) \le \max_{0 \le k \le s-1} \deg(P_k) = \deg(P_{s-1}) < \deg(P_s)$.
        \1 Ce qui est absurde, donc $(P_0,\dots,P_r)$ est une famille libre de $\Kset_n[X]$.
    \end{outline}
\end{exercise}

\begin{example}
    On pose $\forall \alpha \in \R, f_\alpha : \begin{cases} \R \to \R \\ x \mapsto e^{\alpha x} \end{cases}$

    $(f_\alpha)_{\alpha \in \R}$ est une famille libre de $\R^\R$.

    \textbf{Justification :}

    \begin{outline}
        \1 Soit $n \in \N^*$, et soient $\alpha_1,\dots,\alpha_n \in \R$ tels que $\alpha_1<\dots<\alpha_n$.
        \1 On suppose que $(f_{\alpha_k})_{1 \le k \le n}$ n'est pas une famille libre de $\R^\R$.
        \1 Donc $\exists (\lambda_1,\dots,\lambda_n) \in \R^n\setminus\{0_\R^n\} \tq \sum_{k=1}^n \lambda_k f_{\alpha_k} = 0$.
        \1 Donc $\forall x \in \R, \sum_{k=1}^n \lambda_k e^{\alpha_k x} = 0$.
        \1 Soit $s = \max(\{ k \in \{1,\dots,n\} \tq \lambda_k \ne 0\})$.
        \1 Donc $\lambda_s \ne 0$ et $\forall k > s, \lambda_k = 0$.
        \1 Donc $\forall x \in \R, \sum_{k=1}^s \lambda_k e^{\alpha_k x} = 0$.
        \1 Donc $\forall x \in \R, \lambda_s e^{\alpha_s x} = -\sum_{k=1}^{s-1} \lambda_k e^{\alpha_k x}$.
        \1 Donc $\forall x \in \R, \lambda_s = -\sum_{k=1}^{s-1} \lambda_k e^{\underbrace{(\alpha_k - \alpha_s)}_{\le 0} x}$.
        \1 Par passage à la limite $x \to +\infty$, on a $\lambda_s = 0$, ce qui est absurde, donc $(f_{\alpha_k})_{1 \le k \le n}$ est une famille libre de $\cclas\infty(\R,\R)$.
    \end{outline}
\end{example}

\begin{property}
    Toute sous-famille d'une famille libre est une famille libre.

    i.e. si $(e_i)_{i \in I}$ est une famille libre de $E$, et $I' \subset I$, alors $(e_i)_{i \in I'}$ est une famille libre de $E$.
\end{property}

\begin{proof}
    Supposons que $(e_i)_{i \in I}$ est libre et soit $I' \subset I$.
    
    Soit $J \subset I'$ un sous-ensemble fini de $I'$, alors $J \subset I$ et comme $(e_i)_{i \in I}$ est libre, alors $(e_j)_{j \in J}$ est libre, donc $(e_i)_{i \in I'}$ est libre.
\end{proof}

\begin{definition}
    Soient $E$ un $\Kset$-ev. et $(e_i)_{i \in I}$ une famille d'éléments de $E$.

    On dit que $(e_i)_{i \in I}$ est une base de $E$ lorsque :
    \begin{enumerate}
        \item $(e_i)_{i \in I}$ est une famille libre de $E$.
        \item $(e_i)_{i \in I}$ est une famille génératrice de $E$.
    \end{enumerate}
\end{definition}

\begin{example}
    \begin{enumerate}
        \item $(e_1,\dots,e_n)$ est une base de $\Kset^n$.
        \item $(X^k)_{k \in \N}$ est une base de $\Kset[X]$.
        \item $(1,X,\dots,X^n)$ est une base de $\Kset_n[X]$.
        \item Soient $x_0,\dots,x_n \in \Kset$ deux à deux distincts. Alors les polynômes de Lagrange associés à $x_0,\dots,x_n$ forment une base de $\Kset_n[X]$.
        \begin{outline}
            \1 En effet, on sait déjà que $(l_0,\dots,l_n)$ est une famille libre de $\Kset_n[X]$.
            \1 Montrons maintenant qu'il s'agit d'une famille libre.
                \2 Soit $P \in \Kset_n[X]$, alors $\deg(P) \le n$, donc $P$ est déterminé par ses valeurs en $n+1$ points distincts, par exemple $x_0,\dots,x_n$.
                \2 Donc $\forall k \in \{0,\dots,n\}, P(x_k) = \sum_{i=0}^n P(x_i) l_i(x_k) = P(x_k)$, donc $P = \sum_{i=0}^n P(x_i) l_i$, donc $(l_0,\dots,l_n)$ est une famille génératrice de $\Kset_n[X]$.
                \2 Sinon, on peut poser $Q = \sum_{i=0}^n P(x_i) l_i$, calculer sa différence avec $P$, montrer qu'il admet $n+1$ racines distinctes, et conclure que la différence est nulle, donc $P = Q$.
                \2 Donc $\forall P \in \Kset_n[X], P \in \Vect(l_0,\dots,l_n)$, donc $(l_0,\dots,l_n)$ est une famille génératrice de $\Kset_n[X]$.
        \end{outline}
    \end{enumerate}
\end{example}

\begin{theorem}
    Soient $n \in \N^*$, $E$ un $\Kset$-ev. et $e_1,\dots,e_n$ des éléments de $E$.

    Si $v_1,\dots,v_{n+1} \in \Vect(e_1,\dots,e_n)$ alors $(v_1,\dots,v_{n+1})$ est liée.
\end{theorem}

\begin{proof}
    La démonstration se fait par récurrence sur $n$.

    \textbf{Initialisation :} $n=1$
\end{proof}

\begin{consequence}
    Si $(e_1,\dots,e_n)$ est libre dans $E$, un $\Kset$-ev., et $(g_1,\dots,g_{m})$ est une famille génératrice de $E$, alors $n \le m$.
\end{consequence}

\begin{proof}
    Supposons que $n > m$.

    Donc $e_1,\dots,e_n \in \Vect(g_1,\dots,g_m)$, donc $(e_1,\dots,e_n)$ est liée, ce qui est absurde, donc $n \le m$.
\end{proof}

\begin{consequence}
    Soient $E$ un $\Kset$-ev. et $L$ une famille libre, $G$ une famille génératrice et $B$ une base de $E$.

    Alors $\card{L} \le \card{B} \le \card{G}$.
\end{consequence}
